## 1.1 操作系统的定义

操作系统（Operating System，简称 OS）是计算机系统中最基本的系统软件，它管理计算机的硬件和软件资源，为应用程序和用户提供运行环境。没有操作系统，计算机硬件就是一堆无用的电子元件。

### 操作系统的地位

从计算机组成的角度来看，操作系统位于：
- **硬件层之上**：直接控制和管理硬件
- **应用软件之下**：为应用软件提供系统调用接口

> **核心定义**：操作系统是管理和控制计算机软硬件资源的程序集合，是用户与计算机硬件之间的桥梁。

### 进程控制块（PCB）

每个进程都有一个 PCB，用于保存进程的管理信息：

```c
struct PCB {
    int pid;                    // 进程标识符
    int state;                  // 进程状态：就绪/运行/阻塞
    int priority;               // 进程优先级
    unsigned long start_time;   // 开始运行时间
    unsigned long cpu_time;     // 已用CPU时间
    struct file *files[NR_OPEN];// 打开的文件表
    struct mm_struct *mm;       // 进程的内存管理信息
    struct thread_struct *thread;// 寄存器上下文
};
```

### 进程状态转换

```mermaid
stateDiagram-v2

    就绪 --> 运行: 调度程序选中
```

### 实际应用场景：Web 服务器进程管理

当你在浏览器中访问一个网站时：

```
1. 浏览器（客户端进程）发送 HTTP 请求
2. Web 服务器创建子进程处理请求
3. 父进程继续监听新连接
4. 子进程处理完成后被终止
```

```python
# Python 示例：简单的多进程 Web 服务器概念
import os

def handle_request(client_socket):
    """处理客户端请求"""
    request = client_socket.recv(1024)
    response = b"HTTP/1.1 200 OK\r\n\r\nHello, World!"
    client_socket.send(response)
    client_socket.close()

def main():
    """主进程：接受连接并创建子进程处理"""
    server_socket = create_server_socket()
    
    while True:
        client_socket, addr = server_socket.accept()
        pid = os.fork()  # 创建子进程
        
        if pid == 0:
            # 子进程：处理请求
            handle_request(client_socket)
            exit(0)  # 处理完退出
        else:
            # 父进程：关闭客户端套接字，继续监听
            client_socket.close()
            # 父进程继续接受新连接
```

### 代码示例：Linux 内存映射

```c
#include <stdio.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <string.h>

/**
 * 内存映射示例
 * 使用 mmap 系统调用将文件映射到内存
 */
int main() {
    // 创建并打开一个文件
    FILE *fp = fopen("test.dat", "w+");
    if (fp == NULL) {
        perror("fopen");
        return 1;
    }
    
    // 写入数据
    const char *data = "Hello, Operating System!";
    fwrite(data, 1, strlen(data), fp);
    fclose(fp);
    
    // 使用内存映射读取文件
    int fd = open("test.dat", O_RDONLY);
    if (fd == -1) {
        perror("open");
        return 1;
    }
    
    // 将文件映射到内存
    char *mapped = mmap(NULL,              // 让内核选择地址
                       4096,               // 映射大小
                       PROT_READ,          // 可读
                       MAP_PRIVATE,        // 私有映射
                       fd,                 // 文件描述符
                       0);                 // 偏移量
    
    if (mapped == MAP_FAILED) {
        perror("mmap");
        close(fd);
        return 1;
    }
    
    // 读取映射的内存（实际从文件读取）
    printf("Mapped content: %s\n", mapped);
    
    // 解除映射
    munmap(mapped, 4096);
    close(fd);
    
    return 0;
}
```


### 代码示例：Python 文件操作

```python
"""
文件系统操作示例
演示常见的文件读写和目录操作
"""
import os
import shutil

# 1. 创建目录结构
def create_project_structure():
    """创建项目目录结构"""
    dirs = [
        "project/src",
        "project/tests", 
        "project/docs",
        "project/config"
    ]
    
    for d in dirs:
        os.makedirs(d, exist_ok=True)
        print(f"Created: {d}")

# 2. 读取文件内容
def read_file_content(filepath):
    """安全地读取文件内容"""
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except FileNotFoundError:
        print(f"File not found: {filepath}")
        return None
    except PermissionError:
        print(f"Permission denied: {filepath}")
        return None

# 3. 递归列出目录内容
def list_directory_recursive(path, indent=0):
    """递归列出目录内容"""
    try:
        entries = os.listdir(path)
        for entry in entries:
            full_path = os.path.join(path, entry)
            prefix = "  " * indent
            if os.path.isdir(full_path):
                print(f"{prefix}[DIR]  {entry}/")
                list_directory_recursive(full_path, indent + 1)
            else:
                size = os.path.getsize(full_path)
                print(f"{prefix}[FILE] {entry} ({size} bytes)")
    except PermissionError:
        print(f"{'  ' * indent}[Permission denied]")

# 4. 文件元数据获取
def get_file_info(filepath):
    """获取文件详细信息"""
    stat = os.stat(filepath)
    return {
        'size': stat.st_size,
        'created': stat.st_ctime,
        'modified': stat.st_mtime,
        'accessed': stat.st_atime,
        'is_file': os.path.isfile(filepath),
        'is_dir': os.path.isdir(filepath)
    }

if __name__ == "__main__":
    # 创建测试文件
    with open("project/README.txt", "w") as f:
        f.write("# 我的项目\n\n这是一个测试项目。")
    
    print("=== 目录结构 ===")
    list_directory_recursive("project")
    
    print("\n=== 文件信息 ===")
    info = get_file_info("project/README.txt")
    for key, value in info.items():
        print(f"{key}: {value}")
    
    # 清理
    shutil.rmtree("project")
    print("\n项目已清理")
```

```c
                            size_t count, loff_t *offset)
{
    char kernel_buf[256];
    
    if (count > sizeof(kernel_buf) - 1)
        count = sizeof(kernel_buf) - 1;
    
    if (copy_from_user(kernel_buf, buf, count))
        return -EFAULT;
    
    kernel_buf[count] = '\0';
    printk(KERN_INFO "my_device: Received %zu bytes: %s\n", count, kernel_buf);
    
    return count;
}

/* 文件操作结构体 */
static struct file_operations fops = {
    .owner   = THIS_MODULE,
    .open    = device_open,
    .release = device_release,
    .read    = device_read,
    .write   = device_write,
};

/* 模块初始化 */
static int __init my_driver_init(void)
{
    /* 动态分配设备号 */
    if (alloc_chrdev_region(&dev_num, 0, 1, DEVICE_NAME) < 0) {
        printk(KERN_ALERT "my_device: Failed to allocate device number\n");
        return -1;
    }
    
    /* 初始化字符设备 */
    cdev_init(&my_cdev, &fops);
    my_cdev.owner = THIS_MODULE;
    
    /* 添加字符设备到系统 */
    if (cdev_add(&my_cdev, dev_num, 1) < 0) {
        unregister_chrdev_region(dev_num, 1);
        printk(KERN_ALERT "my_device: Failed to add cdev\n");
        return -1;
    }
    
    printk(KERN_INFO "my_device: Driver loaded, major number %d\n", MAJOR(dev_num));
    return 0;
}

/* 模块清理 */
static void __exit my_driver_exit(void)
{
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_num, 1);
    printk(KERN_INFO "my_device: Driver unloaded\n");
}

module_init(my_driver_init);
module_exit(my_driver_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Example");
MODULE_DESCRIPTION("Simple Character Device Driver");
```

### 分层结构的优缺点

| 优点 | 缺点 |
|------|------|
| 模块化清晰 | 层次划分困难 |
| 易于调试和验证 | 效率较低（每层调用都有开销） |
| 便于维护和扩展 | 难以选择最优层次数 |
| 底层可信，可被上层信任 | 紧密耦合的模块难以分层 |

- **VMS**：DEC的操作系统

### 微内核 vs 宏内核

| 特性 | 微内核（Microkernel） | 宏内核（Monolithic） |
|------|----------------------|---------------------|
| 内核大小 | 很小（< 100KB） | 很大（MB级） |
| 服务位置 | 用户空间 | 内核空间 |
| 通信方式 | IPC（消息传递） | 函数调用 |
| 可扩展性 | 高（添加服务不影响内核） | 低（修改内核困难） |
| 性能 | 较差（IPC开销） | 较好（无 IPC 开销） |
| 可靠性 | 高（服务崩溃不影响内核） | 低（内核崩溃导致系统崩溃） |
| 代表系统 | MINIX, QNX, seL4 | Linux, Windows, macOS |

### 代码示例：微内核进程间通信

```c
/**
 * 微内核 IPC 示例
 * 客户端通过消息传递与服务端通信
 */

/* 消息结构 */
typedef struct {
    int src_pid;           // 源进程ID
    int dest_service;      // 目标服务ID
    int msg_type;          // 消息类型
    size_t data_len;       // 数据长度
    char data[256];        // 消息数据
} message_t;

/* 客户端：发送文件读取请求 */
int client_read_file(const char *filename, char *buffer, size_t len) {
    message_t msg;
    
    // 构造消息
    msg.src_pid = get_current_pid();
    msg.dest_service = FILE_SYSTEM_SERVICE;
    msg.msg_type = MSG_READ_FILE;
    msg.data_len = strlen(filename) + 1;
    strcpy(msg.data, filename);
    
    // 发送消息（内核态 IPC）
    int status = send_message(&msg);
    if (status != 0) {
        return -1;  // 发送失败
    }
    
    // 等待响应
    message_t reply;
    receive_message(&reply, FILE_SYSTEM_SERVICE);
    
    // 复制数据到用户缓冲区
    if (reply.msg_type == MSG_READ_RESPONSE) {
        memcpy(buffer, reply.data, len);
        return reply.data_len;
    }
    
    return -1;  // 读取失败
}

/* 服务端：处理文件读取请求 */
void file_server_loop() {
    message_t msg;
    
    while (1) {
        // 等待客户端请求
        receive_message(&msg, FILE_SYSTEM_SERVICE);
        
        if (msg.msg_type == MSG_READ_FILE) {
            // 处理文件读取
            int fd = open(msg.data, O_RDONLY);
            if (fd >= 0) {
                char content[256];
                int n = read(fd, content, sizeof(content));
                close(fd);
                
                // 发送响应
                message_t reply;
                reply.msg_type = MSG_READ_RESPONSE;
                reply.data_len = n;
                memcpy(reply.data, content, n);
                send_message(&reply);
            }
        }
    }
}
```

### 外核设计原则

| 原则 | 说明 |
|------|------|
| 极简内核 | 内核只做资源保护和分配 |
| 硬件直接访问 | 应用或库操作系统直接管理硬件 |
| 灵活性优先 | 让应用自己决定如何抽象资源 |
| 安全隔离 | 内核确保资源访问权限正确 |

### 典型外核系统

- **MIT Exokernel**：外核研究的先驱
- **Xen**：虚拟机监视器（部分采用外核思想）
- **DROPS**：分布式实时系统

### 外核架构代码概念

```c
/**
 * 外核资源分配示例
 * 内核只负责分配和保护物理资源
 */

/* 物理内存页面分配 */
int exokernel_alloc_pages(int pid, int num_pages, page_t **pages) {
    // 检查进程是否有权限分配
    if (!check_permission(pid, RESOURCE_MEMORY, num_pages)) {
        return -EACCES;
    }
    
    // 分配物理页面
    page_t *allocated = physical_alloc(num_pages);
    if (!allocated) {
        return -ENOMEM;
    }
    
    // 设置保护权限
    set_page_protection(allocated, num_pages, PROT_READ | PROT_WRITE);
    
    // 记录资源使用（用于后续保护）
    resource_track(pid, RESOURCE_MEMORY, num_pages);
    
    *pages = allocated;
    return 0;
}

/* 应用程序通过库操作系统使用资源 */
void library_os_page_alloc() {
    page_t *pages;
    
    // 请求外核分配页面
    int ret = sys_page_alloc(&pages, 1);
    if (ret == 0) {
        // 直接访问物理页面
        memset(pages, 0, PAGE_SIZE);
    }
}
```
