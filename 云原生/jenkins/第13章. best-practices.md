# 第13章：Jenkins 最佳实践与优化

本书的最后一章将详细介绍 Jenkins 在企业环境中的最佳实践，涵盖从编写高质量 Pipeline 到架构设计的各个层面。这些最佳实践基于全球数万名 Jenkins 用户的生产经验总结，能够帮助团队构建安全、高效、可靠 CI/CD 系统。

## 13.1 Jenkinsfile 组织与结构最佳实践

Jenkinsfile 是 Pipeline 的核心载体，其组织结构直接影响 Pipeline 的可维护性、可读性和执行效率。本节将深入探讨如何组织 Jenkinsfile，使其符合企业级开发标准。

### 13.1.1 目录结构与模块化设计

在大型项目中，Jenkinsfile 应该采用模块化设计，将常用的步骤封装为共享库（Shared Library），避免重复代码。典型的项目结构如下：

```
project-root/
├── Jenkinsfile                    # 主 Pipeline 文件
├── vars/                          # 共享库自定义步骤
│   ├── buildArtifact.groovy        # 构建制品步骤
│   ├── runTests.groovy             # 运行测试步骤
│   ├── deployToEnv.groovy          # 部署环境步骤
│   └── notifyTeam.groovy           # 通知团队步骤
├── src/                           # 共享库源码
│   └── com/
│       └── myorg/
│           ├── BuildUtils.groovy  # 构建工具类
│           ├── DeployConfig.groovy # 部署配置类
│           └── PipelineStage.groovy # Pipeline 阶段类
├── resources/                      # 共享库资源文件
│   ├── templates/
│   │   └── notification.html
│   └── config/
│       └── environments.yaml
├── ci/
│   ├── build.groovy                # 构建配置
│   ├── test.groovy                 # 测试配置
│   └── deploy.groovy               # 部署配置
└── .jenkins/                      # Jenkins 相关配置
    ├── plugins.txt                # 插件版本锁定
    └── configuration.yml           # Jenkins 配置
```

这种目录结构的核心思想是分离关注点：`vars/` 目录存放简单的闭包（Closure）形式的自定义步骤，而 `src/` 目录存放需要复杂逻辑的类。`resources/` 目录用于存储配置文件和模板文件。

### 13.1.2 主 Jenkinsfile 设计模式

一个结构良好的 Jenkinsfile 应该清晰展示 Pipeline 的流程，同时将具体实现细节委托给共享库。以下是一个参考实现：

```groovy
// Jenkinsfile - 主 Pipeline 文件
// 版本: 1.0.0
// 维护团队: DevOps Team
// 描述: 微服务项目的标准 CI/CD Pipeline

@Library('myorg-shared-library@main') _

import com.myorg.PipelineConfig
import com.myorg.GitUtils

// Pipeline 配置
def config = PipelineConfig.load([
    appName: 'payment-service',
    buildTools: ['maven': '3.9.0', 'nodejs': '18.0.0'],
    environments: ['dev', 'staging', 'production'],
    notification: [
        slack: [channel: '#ci-cd', webhook: 'https://hooks.slack.com/services/xxx'],
        email: [recipients: 'dev-team@myorg.com']
    ]
])

// 主 Pipeline 流程
pipeline {
    agent {
        label config.getBuildAgentLabel()
    }

    options {
        // 超时设置：防止 hung build
        timeout(time: 1, unit: 'HOURS')

        // 构建编号格式
        buildDiscarder(
            logRotator(numToKeepStr: '50', artifactNumToKeepStr: '20')
        )

        // 禁用并发构建同一分支
        disableConcurrentBuilds()

        // 保持 Pipeline 定义的安静期
        quietPeriod(30)

        // 控制台输出时间戳
        timestamps()
    }

    environment {
        // 全局环境变量
        APP_NAME = config.appName
        APP_VERSION = "${env.BUILD_NUMBER}"
        REGISTRY = 'registry.myorg.com'
        DOCKER_IMAGE = "${REGISTRY}/${APP_NAME}:${APP_VERSION}"
        GIT_COMMIT_SHORT = "${GIT_COMMIT[0..7]}"
    }

    parameters {
        choice(
            name: 'TARGET_ENV',
            choices: ['dev', 'staging', 'production'],
            description: '选择部署目标环境'
        )
        booleanParam(
            name: 'SKIP_TESTS',
            defaultValue: false,
            description: '跳过测试阶段（仅用于紧急构建）'
        )
        string(
            name: 'BUILD_ARGS',
            defaultValue: '',
            description: '额外的 Docker 构建参数'
        )
    }

    stages {
        stage('初始化') {
            steps {
                script {
                    // 初始化检查
                    GitUtils.validateCommit(env.GIT_COMMIT)
                    config.validateParameters()
                }
                echo "========================================="
                echo "项目: ${config.appName}"
                echo "版本: ${APP_VERSION}"
                echo "分支: ${BRANCH_NAME}"
                echo "环境: ${params.TARGET_ENV}"
                echo "Commit: ${GIT_COMMIT_SHORT}"
                echo "========================================="
            }
        }

        stage('代码检查') {
            parallel {
                stage('代码质量分析') {
                    when {
                        not { changeRequest() }
                    }
                    steps {
                        script {
                            config.runSonarQubeAnalysis()
                        }
                    }
                }
                stage('安全扫描') {
                    when {
                        not { changeRequest() }
                    }
                    steps {
                        script {
                            config.runSecurityScan()
                        }
                    }
                }
            }
        }

        stage('构建') {
            steps {
                script {
                    def buildSteps = config.getBuildSteps()
                    buildSteps.execute()
                }
            }
        }

        stage('测试') {
            when {
                expression { return params.SKIP_TESTS == false }
            }
            parallel {
                stage('单元测试') {
                    steps {
                        script {
                            config.runUnitTests()
                        }
                    }
                }
                stage('集成测试') {
                    steps {
                        script {
                            config.runIntegrationTests()
                        }
                    }
                }
            }
        }

        stage('构建制品') {
            steps {
                script {
                    config.buildDockerImage()
                    config.pushToRegistry()
                }
            }
            post {
                success {
                    archiveArtifacts artifacts: 'target/*.jar', fingerprint: true
                }
            }
        }

        stage('部署') {
            steps {
                script {
                    config.deployTo(params.TARGET_ENV)
                }
            }
        }

        stage('健康检查') {
            steps {
                script {
                    config.verifyDeployment(params.TARGET_ENV)
                }
            }
        }
    }

    post {
        always {
            script {
                config.cleanup()
            }
        }
        success {
            script {
                config.notifySuccess()
            }
        }
        failure {
            script {
                config.notifyFailure()
            }
        }
        unstable {
            script {
                config.notifyUnstable()
            }
        }
    }
}
```

### 13.1.3 共享库的最佳实践

共享库（Shared Library）是实现代码重用的核心机制。以下是共享库设计的最佳实践：

**目录结构规范：**

```groovy
// vars/目录下的简单步骤 - 使用闭包形式
// vars/buildArtifact.groovy

def call(Map config) {
    // 参数验证
    def requiredParams = ['artifactName', 'buildTool']
    requiredParams.each { param ->
        if (!config.containsKey(param)) {
            error("Missing required parameter: ${param}")
        }
    }

    def artifactName = config.artifactName
    def buildTool = config.buildTool ?: 'maven'
    def classifier = config.classifier ?: ''

    echo "构建制品: ${artifactName}"

    switch (buildTool) {
        case 'maven':
            sh "${config.mavenPath ?: 'mvn'} clean package -DskipTests=false"
            def artifactPath = "target/${artifactName}.jar"
            if (classifier) {
                artifactPath = "target/${artifactName}-${classifier}.jar"
            }
            if (fileExists(artifactPath)) {
                archiveArtifacts artifacts: artifactPath
            }
            break
        case 'gradle':
            sh "${config.gradlePath ?: 'gradle'} build"
            break
        case 'npm':
            sh "npm run build"
            break
        default:
            error("Unsupported build tool: ${buildTool}")
    }

    return this
}
```

**src/目录下的类形式 - 用于复杂逻辑：**

```groovy
// src/com/myorg/PipelineConfig.groovy
package com.myorg

import groovy.transform.ToString
import groovy.transform.builder.Builder

@Builder
@ToString(includeNames = true)
class PipelineConfig {
    String appName
    Map buildTools
    List environments
    Map notification
    String agentLabel
    Map sonarConfig

    static PipelineConfig load(Map<String, Object> config) {
        return PipelineConfig.builder()
            .appName(config.appName ?: 'unknown-app')
            .buildTools(config.buildTools ?: ['maven': '3.8.0'])
            .environments(config.environments ?: ['dev'])
            .notification(config.notification ?: [:])
            .agentLabel(config.agentLabel ?: 'any')
            .sonarConfig(config.sonarConfig ?: [:])
            .build()
    }

    String getBuildAgentLabel() {
        if (agentLabel) return agentLabel

        // 根据应用类型选择合适的 agent
        if (appName.contains('frontend')) {
            return 'nodejs-agent'
        } else if (appName.contains('java') || appName.contains('backend')) {
            return 'java-agent'
        }
        return 'any'
    }

    void validateParameters() {
        if (!appName) {
            throw new IllegalArgumentException("appName is required")
        }
        if (environments.isEmpty()) {
            throw new IllegalStateException("At least one environment must be defined")
        }
    }

    void runSonarQubeAnalysis() {
        echo "执行 SonarQube 代码质量分析..."
        withCredentials([string(credentialsId: 'sonar-token', variable: 'SONAR_TOKEN')]) {
            sh """
                sonar-scanner \
                    -Dsonar.host.url=${sonarConfig.host ?: 'http://sonar:9000'} \
                    -Dsonar.token=${SONAR_TOKEN} \
                    -Dsonar.projectKey=${appName} \
                    -Dsonar.sources=src/main/java \
                    -Dsonar.java.binaries=target/classes
            """
        }
    }
}
```

```groovy
// src/com/myorg/EnvironmentConfig.groovy
package com.myorg

import org.yaml.snakeyaml.Yaml

class EnvironmentConfig {
    private Map<String, Object> environments
    private String currentEnv

    EnvironmentConfig(String configPath, String environment) {
        this.currentEnv = environment
        def yaml = new Yaml()
        def inputStream = new FileInputStream(configPath)
        this.environments = yaml.load(inputStream).environments
    }

    Map getConfig() {
        if (!environments.containsKey(currentEnv)) {
            throw new IllegalStateException("Environment '${currentEnv}' not found in configuration")
        }
        return environments[currentEnv] as Map
    }

    String getDatabaseUrl() {
        def cfg = getConfig()
        def db = cfg.database
        return "jdbc:postgresql://${db.host}:${db.port}/${db.name}"
    }

    int getReplicas() {
        return getConfig().replicas as Integer
    }

    String getMaxMemory() {
        return getConfig().maxMemory
    }
}
```

### 13.2.3 错误处理与异常管理

```groovy
// ============================================
// 错误处理最佳实践
// ============================================

pipeline {
    agent any

    stages {
        stage('执行可能失败的操作') {
            steps {
                script {
                    // 方式1：使用 try-catch 结构
                    try {
                        sh 'docker build -t myapp:1.0 .'
                        echo "Docker 镜像构建成功"
                    } catch (Exception e) {
                        echo "构建失败: ${e.message}"
                        // 可以选择重新抛出异常
                        throw e
                    }

                    // 方式2：使用 result 变量
                    def buildResult = sh(
                        script: 'mvn package -Dmaven.test.failure.ignore=true',
                        returnStatus: true
                    )

                    if (buildResult != 0) {
                        // 手动处理失败情况
                        echo "构建返回非零状态: ${buildResult}"
                        // 更新构建状态为 unstable
                        currentBuild.result = 'UNSTABLE'
                    }

                    // 方式3：使用 input 步骤进行人工确认
                    def proceed = false
                    try {
                        timeout(time: 10, unit: 'MINUTES') {
                            proceed = input(
                                id: 'deploy-confirm',
                                message: '确认部署到生产环境?',
                                ok: '确认部署',
                                submitter: 'devops-team',
                                parameters: [
                                    choice(name: 'ENVIRONMENT', choices: ['dev', 'staging'], description: '选择部署环境')
                                ]
                            )
                        }
                    } catch (org.jenkinsci.plugins.workflow.steps.FlowInterruptedException e) {
                        echo "部署已取消: ${e.message}"
                        return
                    }
                }
            }
        }
    }
}

// ============================================
// 自定义错误处理类
// ============================================

class PipelineException extends Exception {
    private final String stage
    private final String errorCode

    PipelineException(String message, String stage, String errorCode) {
        super(message)
        this.stage = stage
        this.errorCode = errorCode
    }

    String getStage() {
        return stage
    }

    String getErrorCode() {
        return errorCode
    }

    Map toMap() {
        return [
            errorMessage: this.message,
            stage: this.stage,
            errorCode: this.errorCode,
            timestamp: new Date().format('yyyy-MM-dd HH:mm:ss')
        ]
    }
}
```

```groovy
    }

// 自定义日志工具类
import groovy.util.logging.Slf4j

@Slf4j
class BuildLogger implements Serializable {
    def steps
    String context

    BuildLogger(steps, String context) {
        this.steps = steps
        this.context = context
    }

    void info(String message) {
        log.info "[${context}] ${message}"
        steps.echo "[INFO] [${context}] ${message}"
    }

    void debug(String message) {
        log.debug "[${context}] ${message}"
        if (System.getenv('LOG_LEVEL') == 'DEBUG') {
            steps.echo "[DEBUG] [${context}] ${message}"
        }
    }

    void error(String message, Throwable error = null) {
        log.error "[${context}] ${message}"
        steps.echo "[ERROR] [${context}] ${message}"
        if (error) {
            steps.echo "堆栈跟踪: ${error.getMessage()}"
        }
    }

    void metric(String name, Object value) {
        steps.echo "[METRIC] [${context}] ${name}=${value}"
        // 可以发送到监控系统
    }
}
```

## 13.3 Groovy 脚本安全

Jenkins Pipeline 中的 Groovy 脚本可能成为安全风险的来源。本节详细介绍如何编写安全的 Groovy 脚本，防止常见的安全漏洞。

### 13.3.1 常见安全风险与防护

**命令注入防护：**

```groovy
// ============================================
// 命令注入防护示例
// ============================================

pipeline {
    agent any

    stages {
        stage('安全执行命令') {
            steps {
                script {
                    // ❌ 不安全的做法：直接使用用户输入执行命令
                    // 绝对不要这样做！
                    // def userInput = params.USER_INPUT
                    // sh "git checkout ${userInput}"  // 可能被注入恶意命令

                    // ✅ 安全做法1：使用数组形式避免 shell 解析
                    def safeArgs = [
                        'git', 'checkout',
                        params.BRANCH_NAME ?: 'main'
                    ]
                    sh safeArgs.toString()

                    // ✅ 安全做法2：使用 -c 避免字符串拼接
                    sh([script: '''
                        input="''' + params.BRANCH_NAME?.replaceAll("'", "'\\''") + '''"
                        git checkout "$input"
                    '''])

                    // ✅ 安全做法3：严格验证输入格式
                    def branchName = params.BRANCH_NAME
                    if (branchName == null) {
                        branchName = 'main'
                    }
                    // 只允许字母、数字、下划线和连字符
                    if (branchName !=~ /^[a-zA-Z0-9_-]+$/) {
                        error("无效的分支名称: ${branchName}")
                    }
                    sh "git checkout ${branchName}"

                    // ✅ 安全做法4：使用允许列表
                    def allowedBranches = ['main', 'develop', 'release/*']
                    if (branchName in allowedBranches || branchName ==~ /^feature\\/.*$/) {
                        sh "git checkout ${branchName}"
                    } else {
                        error("分支 ${branchName} 不在允许列表中")
                    }
                }
            }
        }
    }
}

// ============================================
// 输入验证工具类
// ============================================

class InputValidator implements Serializable {
    static final String ALLOWED_BRANCH_PATTERN = /^[a-zA-Z0-9_.\/-]+$/
    static final String ALLOWED_TAG_PATTERN = /^[a-zA-Z0-9._-]+$/

    static boolean isValidBranchName(String branchName) {
        if (branchName == null || branchName.isEmpty()) {
            return false
        }
        return branchName ==~ ALLOWED_BRANCH_PATTERN
    }

    static boolean isValidTagName(String tagName) {
        if (tagName == null || tagName.isEmpty()) {
            return false
        }
        return tagName ==~ ALLOWED_TAG_PATTERN
    }

    static String sanitizePath(String path) {
        if (path == null) {
            return ''
        }
        // 移除所有 .. 防止路径遍历攻击
        return path.replaceAll(/\.\./, '')
    }

    static Map validateEnvironment(String env) {
        def allowedEnvs = ['dev', 'staging', 'production']
        if (env in allowedEnvs) {
            return [valid: true, environment: env]
        }
        return [valid: false, error: "Environment '${env}' not allowed"]
    }

    static Map validatePort(String portStr) {
        try {
            int port = portStr.toInteger()
            if (port > 0 && port < 65536) {
                return [valid: true, port: port]
            }
            return [valid: false, error: "Port out of range: ${port}"]
        } catch (NumberFormatException e) {
            return [valid: false, error: "Invalid port number: ${portStr}"]
        }
    }
```

```groovy
// 拓扑排序辅助函数
def topologicalSort(Map graph) {
    def result = []
    def visited = [] as Set
    def tempMark = [] as Set

    def visit = { String node ->
        if (node in tempMark) {
            throw new Exception("检测到循环依赖: ${node}")
        }
        if (node !in visited) {
            tempMark.add(node)
            (graph[node] ?: []).each { dep ->
                visit(dep)
            }
            tempMark.remove(node)
            visited.add(node)
            result.add(0, node)
        }
    }

    graph.keySet().each { visit(it) }
    return result
}

// 按层级分组
def groupByLevel(Map graph) {
    def levels = []
    def placed = [] as Set

    def place = { String node, int level ->
        while (levels.size() <= level) {
            levels.add([])
        }
        if (node !in placed) {
            levels[level].add(node)
            placed.add(node)
        }
    }

    graph.each { node, deps ->
        if (deps.isEmpty()) {
            place(node, 0)
        }
    }

    def maxLevel = 0
    def changed = true
    while (changed) {
        changed = false
        graph.each { node, deps ->
            if (deps.every { it in placed } && node !in placed) {
                def level = deps.collect { d ->
                    levels.findIndexOf { l -> d in l } + 1
                }.max() + 1
                place(node, level)
                maxLevel = Math.max(maxLevel, level)
                changed = true
            }
        }
    }

    return levels
}
```

```groovy
// 性能监控类
class PerformanceMonitor implements Serializable {
    private Map<String, Long> startTimes = [:]
    private Map<String, Long> durations = [:]

    void start(String label) {
        startTimes[label] = System.currentTimeMillis()
    }

    void end(String label) {
        if (startTimes.containsKey(label)) {
            def duration = System.currentTimeMillis() - startTimes[label]
            durations[label] = duration
            startTimes.remove(label)
        }
    }

    void printReport() {
        echo "========== 性能报告 =========="
        durations.sort { it.value }.reverse().each { label, duration ->
            def durationStr = duration > 1000 ?
                String.format("%.2fs", duration / 1000) :
                "${duration}ms"
            echo "${label}: ${durationStr}"
        }
        echo "=============================="
    }

    Map getMetrics() {
        return durations
    }
}
```

### 13.7.1 备份脚本

```groovy
// ============================================
// 完整备份脚本
// ============================================

pipeline {
    agent any

    environment {
        BACKUP_DIR = '/var/jenkins-backup'
        RETENTION_DAYS = 30
        S3_BUCKET = 's3://myorg-jenkins-backup'
        DB_HOST = 'postgres.internal.myorg.com'
        DB_PORT = '5432'

    parameters {
        choice(
            name: 'BACKUP_TYPE',

            description: '选择备份类型'
        )
        booleanParam(
            name: 'ENCRYPT_BACKUP',
            defaultValue: true,
            description: '是否加密备份'
        )
    }

    stages {
        stage('准备备份环境') {
            steps {
                script {
                    def timestamp = new Date().format('yyyy-MM-dd_HH-mm-ss')
                    env.BACKUP_TIMESTAMP = timestamp
                    env.BACKUP_PATH = "${BACKUP_DIR}/backup-${timestamp}"

                    sh """
                        mkdir -p ${BACKUP_PATH}
                        mkdir -p ${BACKUP_DIR}/latest
                    """

                    echo "备份目标目录: ${BACKUP_PATH}"
                }
            }
        }

        stage('备份配置') {
            steps {
                script {
                    echo "开始备份 Jenkins 配置..."

                    // 备份 JENKINS_HOME 目录结构
                    sh """
                        cd \${JENKINS_HOME:-/var/jenkins_home}

                        # 备份配置文件（排除大文件和日志）
                        tar --exclude='./workspace/**' \
                            --exclude='./logs/**' \
                            --exclude='./updates/**' \
                            --exclude='./cache/**' \
                            --exclude='*.log' \
                            -czf ${BACKUP_PATH}/jenkins-config.tar.gz .
                    """

                    // 备份作业配置
                    sh """
                        cd \${JENKINS_HOME:-/var/jenkins_home}/jobs
                        find . -maxdepth 2 -type d -name 'jobs' -exec \
                            tar -czf ${BACKUP_PATH}/jenkins-jobs.tar.gz {} \\;
                    """

                    echo "配置备份完成"
                }
            }
        }

        stage('备份用户数据') {
            steps {
                script {
                    echo "备份用户数据和权限配置..."

                    sh """
                        # 备份用户目录
                        cd \${JENKINS_HOME:-/var/jenkins_home}/users
                        tar -czf ${BACKUP_PATH}/jenkins-users.tar.gz .

                        # 备份 secrets（需要加密存储）
                        cd \${JENKINS_HOME:-/var/jenkins_home}
                        tar -czf ${BACKUP_PATH}/jenkins-secrets.tar.gz secrets/ fingerprints/ plugins.json
                    """

                    echo "用户数据备份完成"
                }
            }
        }

        stage('备份数据库（如果使用）') {
            steps {
                script {
                    try {
                        // PostgreSQL 数据库备份
                        sh """
                            PGPASSWORD=\${POSTGRES_PASSWORD} pg_dump \
                                -h ${DB_HOST} \
                                -p ${DB_PORT} \
                                -U jenkins_user \
                                -d ${DB_NAME} \
                                -Fc \
                                -f ${BACKUP_PATH}/jenkins-db.dump

                            echo "数据库备份完成"
                        """
                    } catch (Exception e) {
                        echo "数据库备份失败（可能未使用数据库）: ${e.message}"
                    }
                }
            }
        }

        stage('备份插件') {
            steps {
                script {
                    echo "备份已安装的插件列表..."

                    sh """
                        cd \${JENKINS_HOME:-/var/jenkins_home}

                        # 导出插件列表和版本
                        ls plugins/ | grep -E '\.jpi$' | while read plugin; do
                            version=\$(unzip -p plugins/\$plugin META-INF/MANIFEST.MF | grep Plugin-Version | cut -d: -f2 | tr -d ' ')
                            echo "\${plugin}:\${version}"
                        done > ${BACKUP_PATH}/plugins.txt

                        # 备份插件文件
                        cp -r plugins/ ${BACKUP_PATH}/plugins/
                    """

                    echo "插件备份完成"
                }
            }
        }

        stage('加密备份（可选）') {
            when {
                expression { return params.ENCRYPT_BACKUP }
            }
            steps {
                script {
                    echo "加密备份文件..."

                    withCredentials([password(scredentialsId: 'backup-encryption-key', variable: 'ENCRYPTION_KEY')]) {
                        sh """
                            # 使用 GPG 加密备份文件
                            for file in ${BACKUP_PATH}/*.tar.gz ${BACKUP_PATH}/*.dump; do
                                if [ -f "\$file" ]; then
                                    gpg --symmetric --batch --yes \
                                        --passphrase "\${ENCRYPTION_KEY}" \
                                        --cipher-algo AES256 \
                                        -o "\$file.gpg" "\$file"
                                    rm "\$file"
                                fi
                            done

                            echo "备份文件加密完成"
                        """
                    }
                }
            }
        }

        stage('上传到云存储') {
            steps {
                script {
                    echo "上传备份到云存储..."

                    sh """
                        aws s3 sync ${BACKUP_PATH}/ ${S3_BUCKET}/backup-${env.BACKUP_TIMESTAMP}/

                        # 更新最新备份符号链接
                        aws s3 sync ${BACKUP_PATH}/ ${S3_BUCKET}/latest/

                        echo "备份上传完成"
                    """
                }
            }
        }

        stage('清理旧备份') {
            steps {
                script {
                    echo "清理超过 ${RETENTION_DAYS} 天的旧备份..."

                    sh """
                        # 清理本地旧备份
                        find ${BACKUP_DIR} -type d -name 'backup-*' -mtime +${RETENTION_DAYS} -exec rm -rf {} \\;

                        # 清理 S3 上的旧备份（保留最近 30 天）
                        aws s3 ls ${S3_BUCKET}/ | while read prefix; do
                            backup_date=\$(echo \$prefix | cut -d'/' -f1 | sed 's/backup-//')
                            if [ -n "\$backup_date" ]; then
                                days_old=\$((\$(date +%s) - \$(date -d "\$backup_date" +%s)) / 86400)
                                if [ \$days_old -gt ${RETENTION_DAYS} ]; then
                                    aws s3 rm --recursive \${S3_BUCKET}/\${prefix}
                                fi
                            fi
                        done

                        echo "旧备份清理完成"
                    """
                }
            }
        }

        stage('验证备份完整性') {
            steps {
                script {
                    echo "验证备份完整性..."

                    sh """
                        cd ${BACKUP_PATH}

                        # 验证 tar 文件完整性
                        for tar_file in *.tar.gz; do
                            if [ -f "\$tar_file" ]; then
                                tar -tzf "\$tar_file" > /dev/null 2>&1
                                if [ \$? -eq 0 ]; then
                                    echo "\${tar_file}: 完整"
                                else
                                    echo "\${tar_file}: 损坏！"
                                    exit 1
                                fi
                            fi
                        done

                        # 计算备份大小
                        du -sh ${BACKUP_PATH}

                        echo "备份验证完成"
                    """
                }
            }
        }
    }

    post {
        always {
            script {
                // 发送备份状态通知
                def backupStatus = currentBuild.result ?: 'SUCCESS'
                def backupSize = sh(script: "du -sh ${BACKUP_PATH} 2>/dev/null | cut -f1", returnStdout: true).trim()

                emailext(
                    subject: "Jenkins 备份 ${backupStatus}: ${env.BACKUP_TIMESTAMP}",
                    body: """
                        备份类型: ${params.BACKUP_TYPE}
                        备份状态: ${backupStatus}
                        备份大小: ${backupSize}
                        备份路径: ${BACKUP_PATH}
                        备份时间: ${new Date().format('yyyy-MM-dd HH:mm:ss')}
                    """,
                    to: 'devops-team@myorg.com'
                )
            }
        }
    }
}
```

```groovy
// 恢复脚本

pipeline {
    agent any

    parameters {
        choice(
            name: 'RESTORE_TYPE',
            choices: ['full', 'config-only', 'jobs-only'],
            description: '选择恢复类型'
        )

        )
        booleanParam(

            defaultValue: true,
            description: '恢复前是否停止 Jenkins'
        )
        booleanParam(
            name: 'DECRYPT_BACKUP',
            defaultValue: true,
            description: '备份文件是否需要解密'
        )
    }

    stages {
        stage('确认恢复操作') {
            steps {
                script {
                    if (params.STOP_JENKINS) {
                        echo "警告：即将停止 Jenkins 服务进行恢复..."
                        input message: '确认执行恢复操作？这将导致 Jenkins 服务短暂中断。', ok: '确认恢复'
                    }
                }
            }
        }

        stage('准备恢复环境') {
            steps {
                script {
                    def backupTimestamp = params.BACKUP_TIMESTAMP ?: 'latest'
                    env.RESTORE_SOURCE = "s3://myorg-jenkins-backup/${backupTimestamp}/"
                    env.RESTORE_DIR = '/var/jenkins-restore'
                    env.JENKINS_HOME = '/var/jenkins_home'

                    sh """
                        mkdir -p ${RESTORE_DIR}
                        mkdir -p ${JENKINS_HOME}
                    """

                    echo "恢复源: ${env.RESTORE_SOURCE}"
                }
            }
        }

        stage('下载备份文件') {
            steps {
                script {
                    echo "从云存储下载备份..."

                    sh """
                        aws s3 sync ${env.RESTORE_SOURCE} ${env.RESTORE_DIR}/

                        echo "下载的备份文件:"
                        ls -la ${env.RESTORE_DIR}/
                    """
                }
            }
        }

        stage('解密备份文件（如果需要）') {
            when {
                expression { return params.DECRYPT_BACKUP }
            }
            steps {
                script {
                    echo "解密备份文件..."

                    withCredentials([password(scredentialsId: 'backup-encryption-key', variable: 'ENCRYPTION_KEY')]) {
                        sh """
                            cd ${env.RESTORE_DIR}

                            for encrypted_file in *.gpg; do
                                if [ -f "\$encrypted_file" ]; then
                                    original_file=\${encrypted_file%.gpg}
                                    gpg --batch --yes \
                                        --passphrase "\${ENCRYPTION_KEY}" \
                                        --decrypt "\$encrypted_file" > "\$original_file"
                                    echo "解密完成: \$original_file"
                                fi
                            done
                        """
                    }
                }
            }
        }

        stage('停止 Jenkins') {
            when {
                expression { return params.STOP_JENKINS }
            }
            steps {
                script {
                    echo "停止 Jenkins 服务..."

                    // 等待正在运行的构建完成（最多等待 10 分钟）
                    sh """
                        # 检查是否有运行的构建
                        BUILD_COUNT=\$(curl -s -u \${JENKINS_API_TOKEN}: ${JENKINS_URL}/api/json?tree=executors\\[currentlyExecutingBuild\\] | grep -c 'number')

                        if [ \$BUILD_COUNT -gt 0 ]; then
                            echo "等待正在运行的构建完成..."
                            # 最多等待 10 分钟
                            for i in \$(seq 1 60); do
                                sleep 10
                                BUILD_COUNT=\$(curl -s -u \${JENKINS_API_TOKEN}: ${JENKINS_URL}/api/json?tree=executors\\[currentlyExecutingBuild\\] | grep -c 'number')
                                [ \$BUILD_COUNT -eq 0 ] && break
                            done
                        fi

                        echo "停止 Jenkins..."
                        systemctl stop jenkins
                    """
                }
            }
        }

        stage('执行恢复') {
            steps {
                script {
                    switch (params.RESTORE_TYPE) {
                        case 'full':
                            echo "执行完整恢复..."
                            sh """
                                # 备份当前配置（以防万一）
                                if [ -d ${JENKINS_HOME} ]; then
                                    cp -r ${JENKINS_HOME} ${JENKINS_HOME}.backup-$(date +%Y%m%d-%H%M%S)
                                fi

                                # 恢复配置文件
                                cd ${env.RESTORE_DIR}
                                tar -xzf jenkins-config.tar.gz -C ${JENKINS_HOME}

                                # 恢复用户数据
                                tar -xzf jenkins-users.tar.gz -C ${JENKINS_HOME}/users

                                # 恢复 secrets（谨慎处理）
                                tar -xzf jenkins-secrets.tar.gz -C ${JENKINS_HOME}

                                # 恢复作业配置
                                tar -xzf jenkins-jobs.tar.gz -C ${JENKINS_HOME}

                                echo "完整恢复完成"
                            """
                            break

                        case 'config-only':
                            echo "恢复配置（保留作业和构建历史）..."
                            sh """
                                cd ${env.RESTORE_DIR}
                                tar -xzf jenkins-config.tar.gz -C ${JENKINS_HOME}
                                echo "配置恢复完成"
                            """
                            break

                        case 'jobs-only':
                            echo "恢复作业配置..."
                            sh """
                                cd ${env.RESTORE_DIR}
                                tar -xzf jenkins-jobs.tar.gz -C ${JENKINS_HOME}
                                echo "作业恢复完成"
                            """
                            break
                    }
                }
            }
        }

        stage('恢复插件') {
            steps {
                script {
                    echo "恢复插件..."

                    sh """
                        cd ${env.RESTORE_DIR}

                        # 读取插件列表
                        if [ -f plugins.txt ]; then
                            while IFS=: read -r plugin version; do
                                if [ -f "plugins/\${plugin}" ]; then
                                    cp plugins/\${plugin} ${JENKINS_HOME}/plugins/\${plugin}
                                    echo "恢复插件: \${plugin} (\${version})"
                                fi
                            done < plugins.txt
                        fi
                    """
                }
            }
        }

        stage('设置权限') {
            steps {
                script {
                    echo "设置正确的文件权限..."

                    sh """
                        # 设置 Jenkins 主目录权限
                        chown -R jenkins:jenkins ${JENKINS_HOME}

                        # 设置关键文件权限
                        chmod 600 ${JENKINS_HOME}/secrets/master.key
                        chmod 600 ${JENKINS_HOME}/hudson.model.User.xml

                        # 设置目录权限
                        chmod -R 755 ${JENKINS_HOME}/jobs
                        chmod -R 755 ${JENKINS_HOME}/workspace

                        echo "权限设置完成"
                    """
                }
            }
        }

        stage('重启 Jenkins') {
            when {
                expression { return params.STOP_JENKINS }
            }
            steps {
                script {
                    echo "重启 Jenkins 服务..."

                    sh """
                        systemctl start jenkins

                        # 等待 Jenkins 启动
                        sleep 30

                        # 验证 Jenkins 状态
                        for i in \$(seq 1 30); do
                            if curl -s http://localhost:8080/login > /dev/null 2>&1; then
                                echo "Jenkins 已成功启动"
                                break
                            fi
                            echo "等待 Jenkins 启动... (\$i/30)"
                            sleep 2
                        done
                    """
                }
            }
        }

        stage('验证恢复结果') {
            steps {
                script {
                    echo "验证恢复结果..."

                    sh """
                        # 检查 Jenkins 健康状态
                        curl -s http://localhost:8080/api/json | jq -r '.mode'

                        # 检查作业数量
                        ls ${JENKINS_HOME}/jobs/ | wc -l

                        # 检查插件状态
                        ls ${JENKINS_HOME}/plugins/*.jpi | wc -l

                        echo "恢复验证完成"
                    """
                }
            }
        }
    }

    post {
        always {
            script {
                def restoreStatus = currentBuild.result ?: 'SUCCESS'

                emailext(
                    subject: "Jenkins 恢复 ${restoreStatus}: ${params.RESTORE_TYPE}",
                    body: """
                        恢复类型: ${params.RESTORE_TYPE}
                        恢复状态: ${restoreStatus}
                        恢复时间: ${new Date().format('yyyy-MM-dd HH:mm:ss')}
                    """,
                    to: 'devops-team@myorg.com'
                )
            }
        }
    }
```

## 13.8 灾难恢复计划

灾难恢复计划（DR Plan）是保障业务连续性的关键。本节详细介绍完整的灾难恢复计划框架。

### 13.8.1 灾难恢复场景

```groovy
// ============================================
// 灾难恢复自动化脚本
// ============================================

pipeline {
    agent any

    environment {
        // DR 站点配置
        DR_JENKINS_URL = 'https://jenkins-dr.internal.myorg.com'
        DR_DB_HOST = 'postgres-dr.internal.myorg.com'
        DR_S3_BUCKET = 's3://myorg-jenkins-backup-dr'
        PRIMARY_JENKINS_URL = 'https://jenkins.internal.myorg.com'
    }

    parameters {
        choice(
            name: 'DR_SCENARIO',
            choices: [
                'master-failure',
                'database-failure',
                'storage-failure',
                'network-failure',
                'full-site-failure'
            ],
            description: '选择灾难场景'
        )
        booleanParam(
            name: 'SKIP_VALIDATION',
            defaultValue: false,
            description: '跳过验证步骤（紧急恢复时使用）'
        )
    }

    stages {
        stage('灾难评估') {
            steps {
                script {
                    echo "========== 灾难评估 =========="
                    echo "灾难场景: ${params.DR_SCENARIO}"
                    echo "发生时间: ${new Date().format('yyyy-MM-dd HH:mm:ss')}"
                    echo "触发人员: ${currentBuild.getBuildCauses()[0]?.getUserId() ?: '系统自动触发'}"

                    // 记录灾难详细信息
                    def drEvent = [
                        scenario: params.DR_SCENARIO,
                        timestamp: new Date().format('yyyy-MM-dd HH:mm:ss'),
                        triggeredBy: currentBuild.getBuildCauses()[0]?.getUserId() ?: 'system'
                    ]

                    // 保存灾难事件记录
                    writeFile file: "${WORKSPACE}/dr-event.json", text: groovy.json.JsonOutput.toJson(drEvent)
                }
            }
        }

        stage('健康检查') {
            when {
                expression { return !params.SKIP_VALIDATION }
            }
            steps {
                script {
                    // 检查主站点状态
                    def primaryHealth = checkJenkinsHealth(env.PRIMARY_JENKINS_URL)
                    echo "主站点健康状态: ${primaryHealth}"

                    // 检查 DR 站点状态
                    def drHealth = checkJenkinsHealth(env.DR_JENKINS_URL)
                    echo "DR 站点健康状态: ${drHealth}"

                    // 检查数据库连接
                    def dbHealth = checkDatabaseHealth()
                    echo "数据库健康状态: ${dbHealth}"

                    // 检查存储可用性
                    def storageHealth = checkStorageHealth()
                    echo "存储健康状态: ${storageHealth}"
                }
            }
        }

        stage('执行恢复') {
            steps {
                script {
                    switch (params.DR_SCENARIO) {
                        case 'master-failure':
                            executeMasterFailover()
                            break
                        case 'database-failure':
                            executeDatabaseFailover()
                            break
                        case 'storage-failure':
                            executeStorageFailover()
                            break
                        case 'network-failure':
                            executeNetworkFailover()
                            break
                        case 'full-site-failure':
                            executeFullSiteFailover()
                            break
                        default:
                            error("未知的灾难场景: ${params.DR_SCENARIO}")
                    }
                }
            }
        }

        stage('验证恢复结果') {
            when {
                expression { return !params.SKIP_VALIDATION }
            }
            steps {
                script {
                    echo "验证恢复结果..."

                    // 验证 Jenkins 服务
                    sh """
                        curl -s ${env.DR_JENKINS_URL}/api/json | jq -r '.mode'
                        curl -s ${env.DR_JENKINS_URL}/api/json | jq -r '.executors'
                    """

                    // 验证关键作业
                    sh """
                        curl -s -u \${JENKINS_API_TOKEN}: ${env.DR_JENKINS_URL}/api/json?tree=jobs[name] | jq '.jobs | length'
                    """

                    // 验证构建历史
                    echo "构建历史验证完成"
                }
            }
        }

        stage('更新 DNS/路由') {
            steps {
                script {
                    echo "更新 DNS 记录..."

                    // 更新 DNS 记录指向 DR 站点
                    // 实际实现取决于 DNS 提供商
                    withCredentials([string(credentialsId: 'dns-api-token', variable: 'DNS_TOKEN')]) {
                        sh """
                            # 示例：更新 Route53 DNS 记录
                            aws route53 change-resource-record-sets \
                                --hosted-zone-id Z1234567890 \
                                --change-batch '{
                                    "Changes": [{
                                        "Action": "UPSERT",
                                        "ResourceRecordSet": {
                                            "Name": "jenkins.internal.myorg.com",
                                            "Type": "CNAME",
                                            "TTL": 300,
                                            "ResourceRecords": [{"Value": "${env.DR_JENKINS_URL}"}]
                                        }
                                    }]'
                        "
                        }
                    }

                    echo "DNS 更新完成"
                }
            }
        }

        stage('通知相关方') {
            steps {
                script {
                    echo "发送灾难恢复通知..."

                    def notification = [
                        scenario: params.DR_SCENARIO,
                        status: currentBuild.result ?: 'SUCCESS',
                        completedAt: new Date().format('yyyy-MM-dd HH:mm:ss'),
                        drUrl: env.DR_JENKINS_URL
                    ]

                    // 发送通知
                    emailext(
                        subject: "[DR 演练] Jenkins 灾难恢复完成",
                        body: groovy.json.JsonOutput.toJson(notification),
                        to: 'devops-team@myorg.com,management@myorg.com'
                    )
                }
            }
        }
    }
}

// ============================================
// 辅助函数
// ============================================

def checkJenkinsHealth(String url) {
    try {
        def response = sh(
            script: "curl -s -o /dev/null -w '%{http_code}' ${url}/api/json",
            returnStdout: true
        ).trim()
        return response == '200' ? 'HEALTHY' : 'UNHEALTHY'
    } catch (Exception e) {
        return "ERROR: ${e.message}"
    }
}

def checkDatabaseHealth() {
    try {
        def result = sh(
            script: """
                PGPASSWORD=\${POSTGRES_PASSWORD} psql \
                    -h ${env.DR_DB_HOST} \
                    -U jenkins_user \
                    -d jenkins \
                    -c 'SELECT 1' \
                    -t
            """,
            returnStdout: true
        ).trim()
        return result == '1' ? 'HEALTHY' : 'UNHEALTHY'
    } catch (Exception e) {
        return "ERROR: ${e.message}"
    }
}

def checkStorageHealth() {
    try {
        def result = sh(
            script: "df -h /var/jenkins_home | tail -1 | awk '{print \$5}'",
            returnStdout: true
        ).trim()
        def usage = result.replace('%', '').toInteger()
        return usage < 90 ? "HEALTHY (${usage}%)" : "WARNING (${usage}%)"
    } catch (Exception e) {
        return "ERROR: ${e.message}"
    }
}

def executeMasterFailover() {
    echo "执行 Master 故障转移..."

    // 1. 检查是否有最新的备份
    def latestBackup = sh(
        script: "aws s3 ls s3://myorg-jenkins-backup/ | sort | tail -1 | awk '{print \$2}'",
        returnStdout: true
    ).trim()

    // 2. 在 DR 站点恢复最新的备份
    sh """
        aws s3 cp --recursive s3://myorg-jenkins-backup/\${latestBackup} /var/jenkins_home/
    """

    // 3. 启动 DR Jenkins 实例
    sh "systemctl start jenkins-dr"
}

def executeDatabaseFailover() {
```

```groovy
// ============================================

// vars/standardPipeline.groovy

def call(Map config = [:]) {
    // 默认配置
    def defaults = [
        appName: 'unknown-app',
        buildTool: 'maven',
        testTypes: ['unit', 'integration'],
        deployEnvironments: ['dev', 'staging'],
        notificationChannels: ['slack'],
        qualityGates: ['test', 'security', 'deploy']
    ]

    // 合并配置
    def mergedConfig = defaults + config

    pipeline {
        agent {
            label mergedConfig.agentLabel ?: 'any'
        }

        options {
            timeout(time: 1, unit: 'HOURS')
            buildDiscarder(logRotator(numToKeepStr: '30'))
            timestamps()
            disableConcurrentBuilds()
        }

        environment {
            APP_NAME = mergedConfig.appName
            BUILD_TOOL = mergedConfig.buildTool
        }

        stages {
            // 代码检查阶段
            stage('代码检查') {
                when { expression { 'code-quality' in mergedConfig.qualityGates } }
                steps {
                    script {
                        mergedConfig.onCodeQualityCheck?.call() ?: echo "跳过代码检查"
                    }
                }
            }

            // 构建阶段
            stage('构建') {
                steps {
                    script {
                        mergedConfig.onBuild?.call() ?: defaultBuild(mergedConfig.buildTool)
                    }
                }
            }

            // 测试阶段
            stage('测试') {
                when { expression { 'test' in mergedConfig.qualityGates } }
                steps {
                    script {
                        mergedConfig.testTypes.each { testType ->
                            stage("测试 - ${testType}") {
                                mergedConfig.onTest?.call(testType) ?: defaultTest(testType)
                            }
                        }
                    }
                }
            }

            // 质量门禁
            stage('质量门禁') {
                when { expression { 'security' in mergedConfig.qualityGates } }
                steps {
                    script {
                        mergedConfig.onQualityGate?.call() ?: echo "执行质量门禁"
                    }
                }
            }

            // 部署阶段
            stage('部署') {
                when { expression { 'deploy' in mergedConfig.qualityGates } }
                steps {
                    script {
                        mergedConfig.onDeploy?.call() ?: echo "准备部署"
                    }
                }
            }
        }

        post {
            always {
                script {
                    mergedConfig.onCleanup?.call() ?: echo "清理资源"
                }
            }
            success {
                script {
                    mergedConfig.onSuccess?.call() ?: echo "构建成功"
                }
            }
            failure {
                script {
                    mergedConfig.onFailure?.call() ?: echo "构建失败"
                }
            }
        }
    }
}

def defaultBuild(String buildTool) {
    switch (buildTool) {
        case 'maven':
            sh 'mvn clean package'
            break
        case 'gradle':
            sh 'gradle build'
            break
        case 'npm':
            sh 'npm run build'
            break
        default:
            error("Unsupported build tool: ${buildTool}")
    }
}

def defaultTest(String testType) {
    echo "执行 ${testType} 测试..."
    // 默认测试逻辑
}
```

```groovy
                <p><span class="label">分支:</span> ${report.changes.branch}</p>
            </div>
        </body>
        </html>
    """.stripIndent()
}
```
