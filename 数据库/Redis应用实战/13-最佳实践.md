## 1. 连接池配置

### 1.1 HikariCP 连接池

HikariCP 是 Spring Boot 默认的数据库连接池，同时也可用于 Redis 连接管理。

```java
import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;
import io.lettuce.core.RedisClient;
import io.lettuce.core.RedisURI;
import io.lettuce.core.resource.ClientResources;
import io.lettuce.core.resource.DefaultClientResources;
import org.apache.commons.pool2.impl.GenericObjectPoolConfig;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.data.redis.connection.RedisConnectionFactory;
import org.springframework.data.redis.connection.RedisStandaloneConfiguration;
import org.springframework.data.redis.connection.lettuce.LettuceConnectionFactory;
import org.springframework.data.redis.connection.lettuce.LettucePoolingClientConfiguration;

import java.time.Duration;

/**
 * Redis 连接池配置类
 * 使用 HikariCP 管理连接池，实现高效的 Redis 连接复用
 */
@Configuration
public class RedisPoolConfig {

    @Value("${spring.data.redis.host:localhost}")
    private String redisHost;

    @Value("${spring.data.redis.port:6379}")
    private int redisPort;

    @Value("${spring.data.redis.password:}")
    private String redisPassword;

    @Value("${spring.data.redis.database:0}")
    private int redisDatabase;

    /**
     * 配置 Lettuce 连接工厂，使用连接池
     */
    @Bean
    public RedisConnectionFactory redisConnectionFactory() {
        // Redis 连接配置
        RedisStandaloneConfiguration serverConfig = new RedisStandaloneConfiguration();
        serverConfig.setHostName(redisHost);
        serverConfig.setPort(redisPort);
        serverConfig.setDatabase(redisDatabase);
        if (redisPassword != null && !redisPassword.isEmpty()) {
            serverConfig.setPassword(redisPassword);
        }

        // 连接池配置 - 使用 HikariCP 风格的参数
        GenericObjectPoolConfig<?> poolConfig = new GenericObjectPoolConfig<>();
        poolConfig.setMaxTotal(50);           // 最大连接数
        poolConfig.setMaxIdle(20);            // 最大空闲连接数
        poolConfig.setMinIdle(5);             // 最小空闲连接数
        poolConfig.setMaxWait(Duration.ofMillis(3000)); // 获取连接最大等待时间
        poolConfig.setTestOnBorrow(true);     // 借用连接时测试有效性
        poolConfig.setTestWhileIdle(true);    // 空闲时测试连接有效性
        poolConfig.setIdleTimeout(Duration.ofSeconds(60)); // 空闲连接超时时间

        // Lettuce 客户端配置
        LettucePoolingClientConfiguration clientConfig = LettucePoolingClientConfiguration.builder()
                .commandTimeout(Duration.ofSeconds(2))  // 命令超时时间
                .poolConfig(poolConfig)
                .build();

        return new LettuceConnectionFactory(serverConfig, clientConfig);
    }
}
```

### 1.2 Lettuce 连接池

Lettuce 是 Spring Boot 3.x 默认的 Redis 客户端，基于 Netty 实现，支持异步和响应式编程。

```java
import io.lettuce.core.ClientOptions;
import io.lettuce.core.SocketOptions;
import io.lettuce.core.cluster.ClusterClientOptions;
import io.lettuce.core.cluster.ClusterTopologyRefreshOptions;
import io.lettuce.core.resource.ClientResources;
import io.lettuce.core.resource.DefaultClientResources;
import org.apache.commons.pool2.impl.GenericObjectPoolConfig;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.data.redis.connection.RedisConnectionFactory;
import org.springframework.data.redis.connection.RedisStandaloneConfiguration;
import org.springframework.data.redis.connection.lettuce.LettuceClientConfiguration;
import org.springframework.data.redis.connection.lettuce.LettuceConnectionFactory;
import org.springframework.data.redis.connection.lettuce.LettucePoolingClientConfiguration;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.data.redis.serializer.GenericJackson2JsonRedisSerializer;
import org.springframework.data.redis.serializer.StringRedisSerializer;

import java.time.Duration;

/**
 * Lettuce 连接池高级配置
 * 包含连接超时、响应超时、连接池调优等配置
 */
@Configuration
public class LettuceAdvancedConfig {

    @Value("${spring.data.redis.host:localhost}")
    private String host;

    @Value("${spring.data.redis.port:6379}")
    private int port;

    @Value("${spring.data.redis.password:}")
    private String password;

    /**
     * 创建客户端资源池（共享资源，推荐）
     * ClientResources 是线程安全的，应该在整个应用生命周期内复用
     */
    @Bean(destroyMethod = "shutdown")
    public ClientResources clientResources() {
        return DefaultClientResources.builder()
                .ioThreadPoolSize(4)   // I/O 线程数
                .computationThreadPoolSize(4)  // 计算线程池大小
                .build();
    }

    /**
     * Socket 连接选项配置
     */
    @Bean
    public SocketOptions socketOptions() {
        return SocketOptions.builder()
                .connectTimeout(Duration.ofSeconds(10))  // 连接超时
                .keepAlive(true)                         // 保持连接活跃
                .tcpNoDelay(true)                        // 禁用 Nagle 算法
                .build();
    }

    /**
     * Lettuce 连接工厂配置
     */
    @Bean
    public RedisConnectionFactory lettuceConnectionFactory(ClientResources clientResources) {
        RedisStandaloneConfiguration serverConfig = new RedisStandaloneConfiguration();
        serverConfig.setHostName(host);
        serverConfig.setPort(port);
        if (password != null && !password.isEmpty()) {
            serverConfig.setPassword(password);
        }

        // 连接池配置
        GenericObjectPoolConfig<?> poolConfig = new GenericObjectPoolConfig<>();
        poolConfig.setMaxTotal(100);        // 最大连接数，根据并发量调整
        poolConfig.setMaxIdle(50);          // 最大空闲连接
        poolConfig.setMinIdle(10);          // 最小空闲连接
        poolConfig.setTestOnBorrow(true);   // 借用时测试
        poolConfig.setTestOnReturn(false);  // 归还时不测试
        poolConfig.setTestWhileIdle(true);   // 空闲时测试
        poolConfig.setTimeBetweenEvictionRuns(Duration.ofMinutes(1)); // 驱逐扫描间隔
        poolConfig.setNumTestsPerEvictionRun(10);  // 每次驱逐测试数量
        poolConfig.setMinEvictableIdleTime(Duration.ofMinutes(5));   // 最小空闲时间

        // 构建 Lettuce 客户端配置
        LettucePoolingClientConfiguration clientConfig = LettucePoolingClientConfiguration.builder()
                .clientResources(clientResources)  // 共享资源
                .commandTimeout(Duration.ofSeconds(5))  // 命令超时
                .poolConfig(poolConfig)
                .build();

        return new LettuceConnectionFactory(serverConfig, clientConfig);
    }

    /**
     * RedisTemplate 配置，支持 JSON 序列化
     */
    @Bean
    public RedisTemplate<String, Object> redisTemplate(RedisConnectionFactory connectionFactory) {
        RedisTemplate<String, Object> template = new RedisTemplate<>();
        template.setConnectionFactory(connectionFactory);

        // 使用 String 序列化器处理键
        StringRedisSerializer stringSerializer = new StringRedisSerializer();
        template.setKeySerializer(stringSerializer);
        template.setHashKeySerializer(stringSerializer);

        // 使用 JSON 序列化器处理值
        GenericJackson2JsonRedisSerializer jsonSerializer = new GenericJackson2JsonRedisSerializer();
        template.setValueSerializer(jsonSerializer);
        template.setHashValueSerializer(jsonSerializer);

        template.afterPropertiesSet();
        return template;
    }
}
```

### 1.3 连接池调优参数

| 参数 | 默认值 | 推荐值 | 说明 |
|------|--------|--------|------|
| `maxTotal` | 8 | 50-100 | 最大连接数，根据并发量调整 |
| `maxIdle` | 8 | 与 maxTotal 相近 | 最大空闲连接数 |
| `minIdle` | 0 | 10-20 | 最小空闲连接数 |
| `maxWait` | -1 | 3000ms | 获取连接最大等待时间 |
| `testOnBorrow` | false | true | 借用时测试连接 |
| `testWhileIdle` | false | true | 空闲时测试连接 |
| `idleTimeout` | 0 | 60000ms | 空闲连接超时 |

```java
/**
 * 连接池调优指南：
 *
 * 1. 估算并发需求：
 *    - 峰值并发数 × 每请求持连接时间 / 响应时间 ≈ 所需连接数
 *    - 例如：1000 QPS，响应时间 100ms，持连时间 100ms
 *    - 所需连接数 = 1000 × 0.1 / 0.1 = 100
 *
 * 2. 监控指标：
 *    - activeCount: 当前活跃连接数
 *    - idleCount: 当前空闲连接数
 *    - waiters: 等待获取连接的线程数
 *
 * 3. 调优步骤：
 *    - 从保守值开始（maxTotal=20）
 *    - 监控连接使用率，如果经常等待或空闲过多，再调整
 *    - 避免设置过大（内存和fd限制）
 */
```

## 2. 键命名规范与过期策略

### 2.1 键命名规范

```java
/**
 * Redis 键命名规范工具类
 * 统一的键命名策略，便于管理和维护
 */
public class RedisKeyRules {

    // 键前缀常量 - 使用冒号分隔层次
    public static final String PREFIX = "app";
    public static final String SEPARATOR = ":";

    // 业务模块前缀
    public static final String MODULE_USER = "user";
    public static final String MODULE_PRODUCT = "product";
    public static final String MODULE_ORDER = "order";
    public static final String MODULE_CACHE = "cache";
    public static final String MODULE_LOCK = "lock";
    public static final String MODULE_RATE_LIMIT = "rate";

    /**
     * 生成用户相关键
     * 格式：app:user:{userId}
     */
    public static String userKey(Long userId) {
        return String.format("%s:%s:%d", PREFIX, MODULE_USER, userId);
    }

    /**
     * 生成用户会话键
     * 格式：app:user:session:{sessionId}
     */
    public static String userSessionKey(String sessionId) {
        return String.format("%s:%s:%s:%s", PREFIX, MODULE_USER, "session", sessionId);
    }

    /**
     * 生成商品缓存键
     * 格式：app:product:{productId}
     */
    public static String productKey(Long productId) {
        return String.format("%s:%s:%d", PREFIX, MODULE_PRODUCT, productId);
    }

    /**
     * 生成商品列表缓存键（带分页）
     * 格式：app:product:list:{categoryId}:{page}:{size}
     */
    public static String productListKey(Long categoryId, int page, int size) {
        return String.format("%s:%s:%s:%d:%d:%d", PREFIX, MODULE_PRODUCT, "list", categoryId, page, size);
    }

    /**
     * 生成订单键
     * 格式：app:order:{orderId}
     */
    public static String orderKey(String orderId) {
        return String.format("%s:%s:%s", PREFIX, MODULE_ORDER, orderId);
    }

    /**
     * 生成分布式锁键
     * 格式：app:lock:{resource}:{identifier}
     */
    public static String lockKey(String resource, String identifier) {
        return String.format("%s:%s:%s:%s", PREFIX, MODULE_LOCK, resource, identifier);
    }

    /**
     * 生成限流键
     * 格式：app:rate:{api}:{ip}
     */
    public static String rateLimitKey(String api, String ip) {
        return String.format("%s:%s:%s:%s", PREFIX, MODULE_RATE_LIMIT, api, ip);
    }

    /**
     * 生成验证码键
     * 格式：app:cache:verify:{bizType}:{bizId}
     */
    public static String verifyCodeKey(String bizType, String bizId) {
        return String.format("%s:%s:%s:%s:%s", PREFIX, MODULE_CACHE, "verify", bizType, bizId);
    }
}
```

### 2.2 过期策略

```java
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.util.concurrent.TimeUnit;

/**
 * Redis 过期策略工具类
 * 提供常用的过期时间设置和 TTL 管理
 */
@Component
public class RedisExpireStrategy {

    private final RedisTemplate<String, Object> redisTemplate;

    public RedisExpireStrategy(RedisTemplate<String, Object> redisTemplate) {
        this.redisTemplate = redisTemplate;
    }

    // ==================== 过期时间常量 ====================

    /** 5分钟 - 短信验证码、临时token */
    public static final Duration EXPIRE_5_MIN = Duration.ofMinutes(5);

    /** 15分钟 - 邮箱验证码、激活链接 */
    public static final Duration EXPIRE_15_MIN = Duration.ofMinutes(15);

    /** 30分钟 - 临时会话、临时缓存 */
    public static final Duration EXPIRE_30_MIN = Duration.ofMinutes(30);

    /** 1小时 - 用户会话 */
    public static final Duration EXPIRE_1_HOUR = Duration.ofHours(1);

    /** 12小时 - 半永久状态 */
    public static final Duration EXPIRE_12_HOUR = Duration.ofHours(12);

    /** 1天 - 日常缓存 */
    public static final Duration EXPIRE_1_DAY = Duration.ofDays(1);

    /** 7天 - 周缓存 */
    public static final Duration EXPIRE_7_DAY = Duration.ofDays(7);

    /** 30天 - 月缓存 */
    public static final Duration EXPIRE_30_DAY = Duration.ofDays(30);

    /**
     * 设置过期时间（通用方法）
     */
    public <T> void set(String key, T value, Duration expire) {
        redisTemplate.opsForValue().set(key, value, expire);
    }

    /**
     * 设置永不过期
     */
    public <T> void setNeverExpire(String key, T value) {
        redisTemplate.opsForValue().set(key, value);
    }

    /**
     * 设置相对过期时间（从现在开始计算）
     */
    public <T> void setRelativeExpire(String key, T value, long amount, TimeUnit unit) {
        redisTemplate.opsForValue().set(key, value, amount, unit);
    }

    /**
     * 更新过期时间
     */
    public Boolean expire(String key, Duration expire) {
        return redisTemplate.expire(key, expire);
    }

    /**
     * 延长过期时间（如果剩余时间少于指定时间才延长）
     */
    public Boolean extendExpireIfNeeded(String key, Duration minRemaining, Duration extendTo) {
        Long ttl = redisTemplate.getExpire(key, TimeUnit.SECONDS);
        if (ttl != null && ttl < minRemaining.getSeconds()) {
            return redisTemplate.expire(key, extendTo);
        }
        return false;
    }

    /**
     * 获取剩余过期时间
     */
    public Long getExpire(String key) {
        return redisTemplate.getExpire(key, TimeUnit.SECONDS);
    }

    /**
     * 判断键是否永不过期
     */
    public boolean isNeverExpire(String key) {
        return getExpire(key) == null || getExpire(key) < 0;
    }

    /**
     * 移除过期时间，设为永不过期
     */
    public Boolean persist(String key) {
        return redisTemplate.persist(key);
    }

    /**
     * 过期策略选择指南：
     *
     * 1. 验证码类（短信、邮箱）：
     *    - 过期时间 = 用户完成验证的合理时间 + 缓冲
     *    - 建议：5-15分钟
     *
     * 2. 会话类（Session、Token）：
     *    - 过期时间 = 用户正常操作周期 × 2
     *    - 建议：1-24小时
     *
     * 3. 缓存类（热点数据）：
     *    - 过期时间 = 数据更新周期 + 缓冲
     *    - 建议：5分钟-1小时
     *
     * 4. 配置类（系统配置）：
     *    - 过期时间 = 配置更新频率
     *    - 建议：1-24小时
     *
     * 5. 排行榜类（Sorted Set）：
     *    - 使用 ZREMRANGEBYSCORE 清理过期数据
     *    - 不建议使用键过期
     */
}
```

### 3.2 淘汰策略

```java
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.data.redis.connection.RedisConnectionFactory;
import org.springframework.data.redis.connection.RedisStandaloneConfiguration;
import org.springframework.data.redis.connection.lettuce.LettuceClientConfiguration;
import org.springframework.data.redis.connection.lettuce.LettuceConnectionFactory;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.data.redis.serializer.StringRedisSerializer;

import io.lettuce.core.ClientOptions;
import io.lettuce.core.resource.ClientOptions.Limit;

/**
 * Redis 内存管理和淘汰策略配置
 */
@Configuration
public class RedisMemoryConfig {

    @Bean
    public LettuceConnectionFactory redisConnectionFactory() {
        RedisStandaloneConfiguration config = new RedisStandaloneConfiguration();

        LettuceClientConfiguration clientConfig = LettuceClientConfiguration.builder()
                .clientOptions(
                        ClientOptions.builder()
                                // 最大连接数限制，防止内存溢出
                                .autoReconnect(true)
                                .build()
                )
                .build();

        return new LettuceConnectionFactory(config, clientConfig);
    }
}

/**
 * 淘汰策略选择指南：
 *
 * | 策略 | 适用场景 |
 * |------|----------|
 * | noeviction | 数据不允许丢失的业务（如配置中心） |
 * | allkeys-lru | 通用缓存，热点数据优先 |
 * | allkeys-lfu | 访问频率差异大的场景 |
 * | allkeys-random | 缓存数据重要性相当 |
 * | volatile-lru | 有明确过期时间的缓存 |
 * | volatile-lfu | 有过期时间且访问频率差异大 |
 * | volatile-random | 有过期时间且数据重要性相当 |
 * | volatile-ttl | 优先淘汰即将过期的数据 |
 *
 * 推荐配置：
 * - 缓存场景：allkeys-lru 或 allkeys-lfu
 * - 会话场景：volatile-lru（会话数据有过期时间）
 */
```

```java
/**
 * Redis 内存优化建议：
 *
 * 1. 键值优化
 *    - 短键名：user:123 比 user:id:123 更节省内存
 *    - 压缩值：对于大字符串使用压缩（zlib/gzip）
 *    - 选择合适数据结构：
 *      - 列表 vs 有序集合（列表更省内存，但功能少）
 *      - 字符串 vs 哈希（批量存储时哈希更省内存）
 *
 * 2. 数据结构选择
 *    String vs Hash:
 *    - String: SET user:1 '{"name":"xxx","age":20}'  // 单个用户信息
 *    - Hash: HSET user:1 name xxx age 20            // 字段可单独操作
 *
 *    当存储多个相似对象时，Hash 更节省内存：
 *    - 1000个用户用String: ~1000 × (key+value) 字节
 *    - 1000个用户用Hash: ~1000 × (key + field数 × field大小) 字节
 *
 * 3. 内存碎片整理
 *    - 定期使用 MEMORY PURGE 命令
 *    - 考虑使用 Redis 内存碎片清理配置
 *
 * 4. 监控内存使用
 *    - INFO memory 查看内存详情
 *    - MEMORY STATS 查看每个键的内存使用
 *    - 使用 SCAN + MEMORY USAGE 分析大键
 */
```

## 4. 主从复制与哨兵

### 4.1 主从复制配置

```yaml
# redis.conf - 主节点配置
bind 0.0.0.0
port 6379
daemonize yes
pidfile /var/run/redis_6379.pid
logfile "redis6379.log"
databases 16

# 设置密码（可选但推荐）
requirepass your_master_password

# 开启 AOF 持久化
appendonly yes
appendfsync everysec
```

```yaml
# redis.conf - 从节点配置
bind 0.0.0.0
port 6380
daemonize yes
pidfile /var/run/redis_6380.pid
logfile "redis6380.log"

# 主节点连接信息
replicaof 127.0.0.1 6379

# 主节点密码
masterauth your_master_password

# 只读模式（从节点默认只读）
replica-read-only yes
```

### 4.2 哨兵模式

```yaml
# sentinel.conf - 哨兵配置
port 26379
daemonize yes
pidfile /var/run/redis-sentinel.pid
logfile "sentinel.log"

# 监控的主节点：名称 IP 端口 投票阈值
sentinel monitor mymaster 127.0.0.1 6379 2

# 主节点密码
sentinel auth-pass mymaster your_master_password

# 主观下线时间（毫秒）
sentinel down-after-milliseconds mymaster 30000

# 故障转移超时时间
sentinel failover-timeout mymaster 180000

# 故障转移时，同时同步新主节点的从节点数量
```

```java
    // ==================== 技巧1：批量操作 ====================

    /**
     * 使用 Pipeline 批量执行命令
     * 减少网络往返次数，提升性能
     */
    public void pipelineExample() {
        // 传统方式：N次网络往返
        for (String key : getKeys()) {
            redisTemplate.opsForValue().get(key);
        }

        // Pipeline 方式：1次网络往返
        redisTemplate.executePipelined((connection) -> {
            for (String key : getKeys()) {
                connection.stringCommands().get(key.getBytes());
            }
            return null;
        });
    }

    /**
     * 使用 MGET 批量获取多个键值
     */
    public List<Object> mgetExample(List<String> keys) {
        if (keys == null || keys.isEmpty()) {
            return Collections.emptyList();
        }
        return redisTemplate.opsForValue().multiGet(keys);
    }

    /**
     * 使用 MSET 批量设置多个键值
     */
    public void msetExample(Map<String, Object> keyValues) {
        if (keyValues == null || keyValues.isEmpty()) {
            return;
        }
        redisTemplate.opsForValue().multiSet(keyValues);
    }

    // ==================== 技巧2：使用 Lua 脚本保证原子性 ====================

    /**
     * 使用 Lua 脚本实现原子操作
     * 避免先 GET 后 SET 的竞态条件
     */
    public String luaScriptExample(String key, long delta) {
        // Lua 脚本：原子递增并返回新值
        String luaScript = """
            local current = redis.call('GET', KEYS[1])
            if current == false then
                current = 0
            else
                current = tonumber(current)
            end
            local newValue = current + tonumber(ARGV[1])
            redis.call('SET', KEYS[1], newValue)
            return newValue
            """;

        return redisTemplate.execute(
                new org.springframework.data.redis.core.script.DefaultRedisScript<>(luaScript, String.class),
                Collections.singletonList(key),
                String.valueOf(delta)
        );
    }

    // ==================== 技巧3：合理使用数据结构 ====================

    /**
     * 使用 HyperLogLog 统计 UV
     * 内存占用极小（12KB），可统计 2^64 个元素
     */
    public void hyperLogLogExample(String date, List<String> userIds) {
        String key = "uv:" + date;
        redisTemplate.opsForHyperLogLog().add(key, userIds.toArray());
        Long size = redisTemplate.opsForHyperLogLog().size(key);
        System.out.println("UV: " + size);

        String key = "sign:" + date;  // 格式：sign:2024-01
```

## 7. 监控指标与告警

```java
import org.springframework.data.redis.connection.RedisConnectionFactory;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import jakarta.annotation.PostConstruct;
import java.lang.management.ManagementFactory;
import java.util.Properties;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicLong;

/**
 * Redis 监控指标收集器
 * 实时采集 Redis 运行指标，支持告警
 */
@Component
public class RedisMetricsCollector {

    private final RedisConnectionFactory connectionFactory;
    private final StringRedisTemplate stringRedisTemplate;

    // 熔断器相关指标
    private final AtomicLong totalRequests = new AtomicLong(0);
    private final AtomicLong failedRequests = new AtomicLong(0);
    private final AtomicLong totalLatency = new AtomicLong(0);

    public RedisMetricsCollector(RedisTemplate<String, Object> redisTemplate,
                                  StringRedisTemplate stringRedisTemplate) {
        this.connectionFactory = redisTemplate.getConnectionFactory();
        this.stringRedisTemplate = stringRedisTemplate;
    }

    @PostConstruct
    public void init() {
        // 启动定时监控任务
        Executors.newSingleThreadScheduledExecutor().scheduleAtFixedRate(
                this::collectAndReport,
                0, 30, TimeUnit.SECONDS
        );
    }

    /**
     * 收集并上报指标
     */
    private void collectAndReport() {
        try {
            Properties info = getRedisInfo();
            if (info == null) return;

            // 收集关键指标
            RedisMetrics metrics = new RedisMetrics();

            // 内存指标
            metrics.usedMemory = parseLong(info.getProperty("used_memory"));
            metrics.maxMemory = parseLong(info.getProperty("maxmemory"));
            metrics.usedMemoryRss = parseLong(info.getProperty("used_memory_rss"));
            metrics.memFragmentationRatio = parseDouble(info.getProperty("mem_fragmentation_ratio"));

            // 连接指标
            metrics.connectedClients = parseInt(info.getProperty("connected_clients"));
            metrics.blockedClients = parseInt(info.getProperty("blocked_clients"));

            // QPS 指标
            metrics.instantaneousOpsPerSec = parseInt(info.getProperty("instantaneous_ops_per_sec"));

            // 持久化指标
            metrics.rdbChangesSinceLastSave = parseLong(info.getProperty("rdb_changes_since_last_save"));
            metrics.aofRewriteInProgress = parseInt(info.getProperty("aof_rewrite_in_progress"));

            // 内存碎片
            metrics.memFragmentationRatio = parseDouble(info.getProperty("mem_fragmentation_ratio"));

            // 打印报告
            printMetricsReport(metrics);

            // 检查告警条件
            checkAlerts(metrics);

        } catch (Exception e) {
            System.err.println("Redis 监控采集失败: " + e.getMessage());
        }
    }

    /**
     * 获取 Redis INFO 信息
     */
    private Properties getRedisInfo() {
        try {
            var connection = connectionFactory.getConnection();
            Properties info = connection.serverCommands().info();
            connection.close();
            return info;
        } catch (Exception e) {
            return null;
        }
    }

    /**
     * 打印指标报告
     */
    private void printMetricsReport(RedisMetrics metrics) {
        System.out.println("========== Redis 监控指标 ==========");
        System.out.println("内存使用: " + formatBytes(metrics.usedMemory) +
                " / " + formatBytes(metrics.maxMemory));
        System.out.println("内存碎片率: " + String.format("%.2f", metrics.memFragmentationRatio));
        System.out.println("连接数: " + metrics.connectedClients +
                " (阻塞: " + metrics.blockedClients + ")");
        System.out.println("QPS: " + metrics.instantaneousOpsPerSec);

        // 计算内存使用率
        if (metrics.maxMemory > 0) {
            double usage = (double) metrics.usedMemory / metrics.maxMemory * 100;
            System.out.println("内存使用率: " + String.format("%.2f%%", usage));
        }

        // 计算请求失败率
        long total = totalRequests.get();
        if (total > 0) {
            long failed = failedRequests.get();
            double failRate = (double) failed / total * 100;
            System.out.println("请求失败率: " + String.format("%.2f%%", failRate));
        }

        // 计算平均延迟
        if (total > 0) {
            long latency = totalLatency.get();
            System.out.println("平均延迟: " + (latency / total) + "ms");
        }

        System.out.println("===================================");
    }

    /**
     * 告警检查
     */
    private void checkAlerts(RedisMetrics metrics) {
        // 内存使用率告警
        if (metrics.maxMemory > 0) {
            double usage = (double) metrics.usedMemory / metrics.maxMemory * 100;
            if (usage > 90) {
                sendAlert("【严重】Redis 内存使用率超过 90%: " + String.format("%.2f%%", usage));
            } else if (usage > 80) {
                sendAlert("【警告】Redis 内存使用率超过 80%: " + String.format("%.2f%%", usage));
            }
        }

        // 内存碎片告警
        if (metrics.memFragmentationRatio > 1.5) {
            sendAlert("【警告】Redis 内存碎片率过高: " + metrics.memFragmentationRatio);
        }

        // 连接数告警
        if (metrics.connectedClients > 10000) {
            sendAlert("【警告】Redis 连接数过多: " + metrics.connectedClients);
        }

        // 阻塞客户端告警
        if (metrics.blockedClients > 0) {
            sendAlert("【警告】Redis 存在阻塞客户端: " + metrics.blockedClients);
        }
    }

    /**
     * 发送告警通知
     */
    private void sendAlert(String message) {
        // TODO: 实现告警通知（钉钉、企业微信、短信等）
        System.err.println("[ALERT] " + message);
    }

    // 辅助方法
    private long parseLong(String value) {
        try {
            return value != null ? Long.parseLong(value) : 0L;
        } catch (NumberFormatException e) {
            return 0L;
        }
    }

    private int parseInt(String value) {
        try {
            return value != null ? Integer.parseInt(value) : 0;
        } catch (NumberFormatException e) {
            return 0;
        }
    }

    private double parseDouble(String value) {
        try {
            return value != null ? Double.parseDouble(value) : 0.0;
        } catch (NumberFormatException e) {
            return 0.0;
        }
    }

    private String formatBytes(long bytes) {
        if (bytes < 1024) return bytes + " B";
        if (bytes < 1024 * 1024) return String.format("%.2f KB", bytes / 1024.0);
        if (bytes < 1024 * 1024 * 1024) return String.format("%.2f MB", bytes / (1024.0 * 1024));
        return String.format("%.2f GB", bytes / (1024.0 * 1024 * 1024));
    }
```

```java
    public void troubleshootOOM() {
        // 使用 SCAN 查找大键
        Set<String> keys = redisTemplate.scan(
                org.springframework.data.redis.core.ScanOptions.scanOptions()
                        .count(1000)
                        .build()
        );

        Map<String, Long> keySizes = new HashMap<>();
        for (String key : keys) {
            Long size = getKeyMemorySize(key);
            if (size != null && size > 1024 * 1024) { // 大于 1MB
                keySizes.put(key, size);
            }
        }

        // 按大小排序输出
        keySizes.entrySet().stream()
                .sorted(Map.Entry.<String, Long>comparingByValue().reversed())
                .limit(10)
                .forEach(e -> log.warn("大键: {} = {} bytes", e.getKey(), e.getValue()));
    }

    private Long getKeyMemorySize(String key) {
        try {
            return redisTemplate.execute((org.springframework.data.redis.core.RedisCallback<Long>) connection -> {
```
